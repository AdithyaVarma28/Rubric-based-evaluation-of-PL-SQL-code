create or replace function incriment_sal(d_no int,n int)
returns varchar as
$$
declare
sal_cu cursor for select sal from emp_deept12 where dno=d_no;
c_sal emp_deept12.sal%type;
avg numeric (10,3);
total numeric(10,3);
m int;
begin
total:=0.000;
select count(eno) into m from emp_deept12 where dno=d_no ;
open sal_cu;
fetch sal_cu into c_sal;
while found
loop
total:=total+c_sal;
avg:=(total/m)*n/100;
update emp_deept12 set sal=sal+avg where d_no=dno;
fetch next from sal_cu into c_sal;
end loop;
return 'updated';
end;
$$
language'plpgsql';
